cmake_minimum_required(VERSION 3.8)
project(piper_moveit_guard)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
add_executable(moveit_guard src/moveit_guard.cpp)
ament_target_dependencies(moveit_guard rclcpp moveit_core moveit_ros_planning moveit_msgs sensor_msgs std_msgs std_srvs visualization_msgs)
target_compile_options(moveit_guard PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS moveit_guard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
